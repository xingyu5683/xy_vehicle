关键修复

def compute_angles_lengths_2D(vector: torch.Tensor, eps: float = 1e-6):
    length = torch.norm(vector, dim=-1)
    # 添加epsilon避免 atan2(0, 0) 产生 NaN
    safe_x = vector[..., 0] + eps * (length < eps).float()
    theta = torch.atan2(vector[..., 1], safe_x)
    return length, theta

🧱 layers_exportable.py - ONNX兼容层

TwoLayerMLP

class TwoLayerMLP(nn.Module):
    """标准两层MLP: Linear → LayerNorm → ReLU → Linear"""
    def __init__(self, input_dim, hidden_dim, output_dim):
        self.mlp = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.LayerNorm(hidden_dim),
            nn.ReLU(inplace=True),
            nn.Linear(hidden_dim, output_dim),
        )

GraphAttentionExportable

核心的图注意力层，完全摆脱 PyTorch Geometric 依赖。

class GraphAttentionExportable(nn.Module):
    """
    ONNX兼容的稀疏图注意力
    
    关键：使用 scatter_add 而非 PyG 的 MessagePassing
    """
    def __init__(self, hidden_dim, num_heads, dropout, has_edge_attr, if_self_attention):
        # Q, K, V 投影
        self.q = nn.Linear(hidden_dim, hidden_dim)
        self.k = nn.Linear(hidden_dim, hidden_dim)
        self.v = nn.Linear(hidden_dim, hidden_dim)
        
        # 边特征投影
        if has_edge_attr:
            self.edge_k = nn.Linear(hidden_dim, hidden_dim)
            self.edge_v = nn.Linear(hidden_dim, hidden_dim)
        
        # FFN
        self.ffn = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim * 4),
            nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(hidden_dim * 4, hidden_dim),
        )

scatter_softmax 实现（ONNX兼容）

def _scatter_softmax(self, src, index, num_nodes):
    """按组计算softmax，使用clamp代替max确保ONNX兼容"""
    src_clipped = torch.clamp(src, min=-50.0, max=50.0)  # 数值稳定性
    exp_src = torch.exp(src_clipped)
    
    # scatter_add 聚合同组的exp值
    sum_exp = torch.zeros(num_nodes, H, device=src.device)
    sum_exp = sum_exp.scatter_add(0, index.unsqueeze(-1).expand(-1, H), exp_src)
    sum_exp_per_edge = sum_exp[index]
    
    return exp_src / (sum_exp_per_edge + 1e-8)

3. 边 (Edge) 计算详解

三种边类型:

边类型 索引格式 属性维度 含义
k2k_t agent_major: a*T+t 6 同一 agent 不同时间步
g2k agent_major: a*T+t 6 polygon → agent
k2k_a time_major: t*N+a 5 同一时间步不同 agent

边属性计算:

// 转换到目标节点的局部坐标系
float dx = src_x - dst_x;
float dy = src_y - dst_y;

// 旋转到 dst 的局部坐标
float cos_h = cos(-dst_heading);
float sin_h = sin(-dst_heading);
float local_x = dx * cos_h - dy * sin_h;
float local_y = dx * sin_h + dy * cos_h;

// 计算边属性
float length = sqrt(local_x² + local_y²) + 1e-8;
float theta = atan2(local_y, local_x);
float heading_diff = wrapAngle(src_heading - dst_heading);

// k2k_t: [length, cos(θ), sin(θ), cos(Δh), sin(Δh), Δt]
// g2k:   [length, cos(θ), sin(θ), cos(Δh), sin(Δh), valid]
// k2k_a: [length, cos(θ), sin(θ), cos(Δh), sin(Δh)]

4. Token 解码与坐标变换

// 1. argmax 获取 token ID
int token_id = argmax(logits, agent_idx * 1024, 1024);

// 2. 查表获取局部位移
TokenEntry delta = token_dict[agent_type][token_id];
// delta = {delta_x, delta_y, delta_heading}

// 3. 局部坐标 → 全局坐标
float heading = current_headings[a];
float cos_h = cos(heading);
float sin_h = sin(heading);

// 旋转矩阵: R(θ) * [dx, dy]^T
float dx_global = delta.delta_x * cos_h - delta.delta_y * sin_h;
float dy_global = delta.delta_x * sin_h + delta.delta_y * cos_h;

// 4. 累加更新
position_x += dx_global;
position_y += dy_global;
heading = wrapAngle(heading + delta.delta_heading);

🔑 关键设计决策

1. 为什么拆分成两个 ONNX？

原因 说明
自回归循环 Python for 无法直接转换为 ONNX Loop 算子
PyG 依赖 MessagePassing, radius_graph 等动态图操作不兼容
条件分支 数据依赖的 if 语句在 trace 时被固化

2. 为什么用 opset 16？

opset 14/15/17 的 ScatterElements 不正确支持 reduction='add'
opset 16 正确实现了这一功能

3. Sink 节点 Padding 策略

无效边索引使用 sink 节点（最后一个索引）而非 0，避免污染真实节点的聚合结果：

SINK_POLYGON = MAX_POLYGONS - 1   # 144
SINK_POLYLINE = MAX_POLYLINES - 1  # 1199

4. TensorRT scatter_add 限制及解决方案

✅ 已解决： 通过自定义 CUDA 插件 libscatter_add_plugin.so 成功支持 TensorRT 部署

问题背景

本模型的图注意力层使用了 scatter_add 操作进行消息聚合：

# PyTorch 代码
sum_exp = sum_exp.scatter_add(0, index, exp_src)

# 对应 ONNX 算子
ScatterElements(axis=0, reduction="add")

TensorRT 原生限制：

  • ScatterElements 仅支持 reduction="none"（覆盖模式）
  • 不支持 reduction="add"（累加模式）

解决方案

方案 状态 说明
TensorRT 自定义插件 ✅ 已实现 libscatter_add_plugin.so，基于 CUDA atomicAdd
ONNX Runtime + CUDA 备选 完全支持，但性能低于 TensorRT FP16
重写模型避免 scatter_add 不推荐 需要改变图注意力架构

自定义插件实现细节

// CUDA kernel 核心逻辑
__global__ void scatter_add_fp16_kernel(
    const __half* src,      // [E, D] 源数据
    const int64_t* index,   // [E, D] 目标索引
    __half* output,         // [N, D] 输出 (原子累加)
    int E, int D, int N
) {
    int e_idx = blockIdx.x * blockDim.x + threadIdx.x;
    int d_idx = blockIdx.y * blockDim.y + threadIdx.y;
    
    if (e_idx < E && d_idx < D) {
        int dst_idx = index[e_idx * D + d_idx];
        if (dst_idx >= 0 && dst_idx < N) {
            atomicAdd(&output[dst_idx * D + d_idx], src[e_idx * D + d_idx]);
        }
    }
}
